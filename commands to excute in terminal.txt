# create venv
python -m venv .venv

# activate venv (PowerShell)
.\\.venv\Scripts\Activate.ps1

# (optional) if activation is blocked by execution policy, run:
Set-ExecutionPolicy -Scope Process -ExecutionPolicy Bypass

# upgrade pip / essentials
python -m pip install --upgrade pip setuptools wheel

# install from requirements.txt
python -m pip install -r .\requirements.txt

# verify installed packages (brief)
python -m pip list

# save installed versions
python -m pip freeze > installed-versions.txt
 

# upload files to git repository
echo "# fake-news-detector" >> README.md
git init
git add README.md
git commit -m "first commit"
git branch -M main
git remote add origin https://github.com/anirudhkowluri/fake-news-detector.git
git push -u origin main
